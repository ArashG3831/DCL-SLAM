cmake_minimum_required(VERSION 3.8)
project(my_epuck_frontier_candidates)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(frontier_exploration_ros2 REQUIRED)
find_package(my_epuck_interfaces REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(webots_ros2_driver REQUIRED)
find_library(WEBOTS_CONTROLLER_LIBRARY Controller
  PATHS /opt/ros/jazzy/lib/controller REQUIRED)
add_library(frontier_search_core src/candidate_utils.cpp)
target_include_directories(frontier_search_core PUBLIC include)
target_link_libraries(frontier_search_core frontier_exploration_ros2::frontier_exploration_ros2_core)
ament_target_dependencies(frontier_search_core geometry_msgs nav_msgs)
add_executable(frontier_candidate_generator src/frontier_candidate_generator.cpp)
target_include_directories(frontier_candidate_generator PRIVATE include)
target_link_libraries(frontier_candidate_generator frontier_search_core)
ament_target_dependencies(frontier_candidate_generator geometry_msgs my_epuck_interfaces nav2_msgs nav_msgs rclcpp rclcpp_action tf2 tf2_ros visualization_msgs)
add_library(chunked_lidar_plugin SHARED src/chunked_lidar_plugin.cpp)
target_include_directories(chunked_lidar_plugin PRIVATE include)
ament_target_dependencies(chunked_lidar_plugin my_epuck_interfaces pluginlib rclcpp webots_ros2_driver)
target_link_libraries(chunked_lidar_plugin ${webots_ros2_driver_LIBRARIES}
  ${WEBOTS_CONTROLLER_LIBRARY})
install(TARGETS frontier_search_core frontier_candidate_generator chunked_lidar_plugin DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY LICENSES DESTINATION share/${PROJECT_NAME})
install(FILES THIRD_PARTY.md DESTINATION share/${PROJECT_NAME})
install(FILES plugin_description.xml DESTINATION share/${PROJECT_NAME})
pluginlib_export_plugin_description_file(webots_ros2_driver plugin_description.xml)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_frontier_core test/test_frontier_core.cpp)
 target_include_directories(test_frontier_core PRIVATE include)
 target_link_libraries(test_frontier_core frontier_search_core)
 ament_target_dependencies(test_frontier_core geometry_msgs nav_msgs)
 ament_add_gtest(test_query_executor_liveness test/test_query_executor_liveness.cpp)
 ament_target_dependencies(test_query_executor_liveness nav2_msgs rclcpp rclcpp_action)
endif()
ament_package()
